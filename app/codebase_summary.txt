Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await _startFirebase();
  runApp(const ProviderScope(child: ClassSyncApp()));
}

/// On Android, Firebase may already be running from google-services.json.
/// Starting it twice throws, so only start it when it isn't running yet.
Future<void> _startFirebase() async {
  try {
    if (Firebase.apps.isEmpty) {
      await Firebase.initializeApp(
        options: DefaultFirebaseOptions.currentPlatform,
      );
    }
  } on FirebaseException catch (error) {
    if (error.code != 'duplicate-app') {
      rethrow;
    }
  }
}

final routerProvider = Provider<GoRouter>((ref) {
  final refresh = ValueNotifier<int>(0);
  ref.onDispose(refresh.dispose);
  ref.listen(authStateProvider, (_, _) => refresh.value++);

  final router = GoRouter(
    initialLocation: AppRoutes.splash,
    refreshListenable: refresh,
    redirect: (context, state) {
      final auth = ref.read(authStateProvider);
      final at = state.matchedLocation;

      if (auth.isLoading) {
        return at == AppRoutes.splash ? null : AppRoutes.splash;
      }

      final signedIn = switch (auth) {
        AsyncData(:final value) => value != null,
        _ => false,
      };

      if (!signedIn) {
        return at == AppRoutes.signIn ? null : AppRoutes.signIn;
      }
      if (at == AppRoutes.signIn || at == AppRoutes.splash) {
        return AppRoutes.timetable;
      }
      return null;
    },
    routes: <RouteBase>[
      GoRoute(
        path: AppRoutes.splash,
        builder: (context, state) => const SplashScreen(),
      ),
      GoRoute(
        path: AppRoutes.signIn,
        builder: (context, state) => const SignInScreen(),
      ),
      GoRoute(
        path: AppRoutes.classForm,
        builder: (context, state) =>
            ClassFormScreen(classId: state.uri.queryParameters['id']),
      ),
      GoRoute(
        path: AppRoutes.friendSchedule,
        builder: (context, state) =>
            FriendScheduleScreen(friendId: state.pathParameters['id']!),
      ),
      StatefulShellRoute.indexedStack(
        builder: (context, state, shell) => AppShell(shell: shell),
        branches: [
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: AppRoutes.timetable,
                builder: (context, state) => const TimetableScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: AppRoutes.availability,
                builder: (context, state) =>
                    const PlaceholderScreen(title: 'Free time screen'),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: AppRoutes.friends,
                builder: (context, state) => const FriendsScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: AppRoutes.settings,
                builder: (context, state) => const MeScreen(),
              ),
            ],
          ),
        ],
      ),
    ],
  );

  ref.onDispose(router.dispose);
  return router;
});

    ],
  );

  @override
  AppColors copyWith({
    Color? muted,
    Color? soft,
    Color? line,
    Color? accent,
    List<Color>? classColors,
  }) {
    return AppColors(
      muted: muted ?? this.muted,
      soft: soft ?? this.soft,
      line: line ?? this.line,
      accent: accent ?? this.accent,
      classColors: classColors ?? this.classColors,
    );
  }

  // We never animate between light and dark, so just switch at the halfway point.
  @override
  AppColors lerp(AppColors? other, double t) {
    if (other == null) return this;
    return t < 0.5 ? this : other;
  }
}

extension AppColorsContext on BuildContext {
  AppColors get appColors => Theme.of(this).extension<AppColors>()!;
}

      onPrimary: Colors.white,
      secondary: Color(0xFFF0A55B),

      surface: Colors.white,
      onSurface: Color(0xFF1E2B28),
    ),
    AppColors.light,
    const Color(0xFFF3F7F5),
  );

  static final ThemeData dark = _build(
    const ColorScheme(
      brightness: Brightness.dark,
      primary: Color(0xFF5BBBA6),
      onPrimary: Color(0xFF0E1513),
      secondary: Color(0xFFF0A55B),
      onSecondary: Color(0xFF2A1A05),
      error: Color(0xFFF0776D),

      surface: Color(0xFF1B2523),
      onSurface: Color(0xFFE8F0ED),
    ),
    AppColors.dark,
    const Color(0xFF111917),
  );

  static ThemeData _build(
    ColorScheme scheme,
    AppColors colors,
    Color background,
  ) {
    final base = ThemeData(
      useMaterial3: true,
      brightness: scheme.brightness,
      colorScheme: scheme,

      displayColor: scheme.onSurface,

/// Turns minutes since midnight into H:MM, so 510 becomes 8:30.
/// 1440 is allowed and shows as 24:00, for classes that end at midnight.
String formatMinutes(int minutes) {
  final safe = minutes < 0 ? 0 : (minutes > 1440 ? 1440 : minutes);
  final mins = (safe % 60).toString().padLeft(2, '0');
  return '${safe ~/ 60}:$mins';
}

/// Seven weekday chips, M to S, in one row. Each takes an equal share of
/// the width, so they never wrap onto a second line.
class DayChips extends StatelessWidget {
  const DayChips({required this.isSelected, required this.onTap, super.key});

  final bool Function(int day) isSelected;
  final ValueChanged<int> onTap;

  @override
  Widget build(BuildContext context) {
    return Row(
      children: [
        for (var day = 1; day <= 7; day++)
          Expanded(
            child: Padding(
              padding: const EdgeInsets.symmetric(horizontal: 2),
              child: _chip(context, day),
            ),
          ),
      ],
    );
  }

  Widget _chip(BuildContext context, int day) {
    final scheme = Theme.of(context).colorScheme;
    final selected = isSelected(day);
    return Semantics(
      button: true,
      selected: selected,
      label: weekdayName(day),
      onTap: () => onTap(day),
      excludeSemantics: true,
      child: Material(
        color: selected ? scheme.primary : scheme.surface,
        shape: StadiumBorder(
          side: BorderSide(
            color: selected ? scheme.primary : context.appColors.line,
          ),
        ),
        child: InkWell(
          customBorder: const StadiumBorder(),
          onTap: () => onTap(day),
          child: SizedBox(
            height: 40,
            child: Center(
              child: Text(
                weekdayInitial(day),
                style: TextStyle(
                  color: selected ? scheme.onPrimary : scheme.onSurface,
                  fontWeight: selected ? FontWeight.w800 : FontWeight.w600,
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}

/// "Maya Reyes" becomes "MR". One word gives one letter.
String initialsOf(String name) {
  final parts = name
      .trim()
      .split(RegExp(r'\s+'))
      .where((p) => p.isNotEmpty)
      .toList();
  if (parts.isEmpty) {
    return '?';
  }
  if (parts.length == 1) {
    return parts.first[0].toUpperCase();
  }
  return (parts.first[0] + parts.last[0]).toUpperCase();
}

import 'dart:async';

/// Keeps a value in memory and tells listeners when it changes.
class FakeStore<T> {
  FakeStore(this._value);

  T _value;
  final _controller = StreamController<T>.broadcast();

  T get value => _value;

  set value(T next) {
    _value = next;
    _controller.add(next);
  }

  /// Emits the current value first, then every change.
  Stream<T> watch() async* {
    yield _value;
    yield* _controller.stream;
  }
}

class FirebaseAuthRepository implements AuthRepository {
  FirebaseAuthRepository({this.firebaseAuth, this.googleSignIn});

  final FirebaseAuth? firebaseAuth;
  final GoogleSignIn? googleSignIn;

  FirebaseAuth get _auth => firebaseAuth ?? FirebaseAuth.instance;

  GoogleSignIn get _google => googleSignIn ?? GoogleSignIn.instance;

  Future<void>? _googleInit;

  Future<void> _ensureGoogleInitialized() {
    return _googleInit ??= _google.initialize();
  }

  @override
  Stream<UserProfile?> watchCurrentUser() {
    return _auth.authStateChanges().asyncExpand((user) {
      if (user == null) {
        return Stream.value(null);
      }

      return FirebaseFirestore.instance
          .collection('users')
          .doc(user.uid)
          .snapshots()
          .map((snapshot) {
            final savedMinutes = snapshot.data()?['defaultReminderMinutes'];
            final defaultReminder = savedMinutes is int ? savedMinutes : 10;

            return _toProfile(
              user,
              defaultReminderMinutes: defaultReminder,
            );
          });
    });
  }

  @override
  Future<UserProfile> signInWithGoogle() async {
    try {
      debugPrint('[Google Sign-In] Initializing Google Sign-In...');

      await _ensureGoogleInitialized();

      debugPrint('[Google Sign-In] Opening account selection...');

      final googleUser = await _google.authenticate();

      debugPrint('[Google Sign-In] Account selected: ${googleUser.email}');

      final idToken = googleUser.authentication.idToken;

      if (idToken == null) {
        debugPrint('[Google Sign-In] ERROR: ID token is null.');

        throw const UnknownFailure('Google did not return a sign-in token.');
      }

      debugPrint(
        '[Google Sign-In] ID token received. '
        'Authenticating with Firebase...',
      );

      final credential = GoogleAuthProvider.credential(idToken: idToken);

      final result = await _auth.signInWithCredential(credential);

      final user = result.user;

      if (user == null) {
        debugPrint('[Google Sign-In] ERROR: Firebase returned no user.');

        throw const UnknownFailure('Sign-in failed. Please try again.');
      }

      debugPrint('[Google Sign-In] SUCCESS: Firebase user authenticated.');

      return _toProfile(user);
    } on AppFailure {
      rethrow;
    } on GoogleSignInException catch (e, stackTrace) {
      debugPrint('[Google Sign-In] ERROR CODE: ${e.code}');
      debugPrint('[Google Sign-In] ERROR DETAILS: $e');
      debugPrintStack(stackTrace: stackTrace);

      if (e.code == GoogleSignInExceptionCode.canceled) {
        throw const NotSignedIn();
      }

      rethrow;
    } on FirebaseAuthException catch (e, stackTrace) {
      debugPrint('[Firebase Auth] ERROR CODE: ${e.code}');
      debugPrint('[Firebase Auth] ERROR MESSAGE: ${e.message}');
      debugPrintStack(stackTrace: stackTrace);

      throw _mapFirebaseError(e);

          id: doc.id,
          name: (data['name'] as String?) ?? '',

/// Live implementation of [FriendRepository] backed by Firebase Firestore.
class FirestoreFriendRepository implements FriendRepository {
  FirestoreFriendRepository(this._firestore, this._userId);

  final FirebaseFirestore _firestore;
  final String _userId;

  CollectionReference get _friendsRef =>
      _firestore.collection('users').doc(_userId).collection('friends');

  CollectionReference get _requestsRef =>
      _firestore.collection('users').doc(_userId).collection('friend_requests');

  @override
  Stream<List<Friend>> watchFriends() {
    return _friendsRef.snapshots().map((snap) {
      return snap.docs.map<Friend>((doc) {
        final data = doc.data() as Map<String, dynamic>;
        final levelStr = data['sharingLevel'] as String? ?? 'busyOnly';
        return Friend(
          friendId: doc.id,
          displayName: (data['displayName'] as String?) ?? 'Friend',
          sharingLevel: levelStr == 'full'
              ? SharingLevel.full
              : SharingLevel.busyOnly,
        );
      }).toList();
    });
  }

  @override
  Stream<List<FriendRequest>> watchIncomingRequests() {
    return _requestsRef.snapshots().map((snap) {
      return snap.docs.map<FriendRequest>((doc) {
        final data = doc.data() as Map<String, dynamic>;
        return FriendRequest(
          id: doc.id,
          fromId: (data['fromUserId'] as String?) ?? '',
          fromName: (data['fromName'] as String?) ?? 'Student',
        );
      }).toList();
    });
  }

  @override
  Future<void> sendRequestByCode(String code) async {
    final query = await _firestore
        .collection('users')
        .where('inviteCode', isEqualTo: code)
        .limit(1)
        .get();

    if (query.docs.isEmpty) {
      throw Exception('Invalid invite code. No user found.');
    }

    final targetUserId = query.docs.first.id;

    if (targetUserId == _userId) {
      throw Exception('You cannot add yourself as a friend.');
    }

    final myDoc = await _firestore.collection('users').doc(_userId).get();
    final myData = myDoc.data() ?? {};
    final myName = (myData['displayName'] as String?) ?? 'Student';

    await _firestore
        .collection('users')
        .doc(targetUserId)
        .collection('friend_requests')
        .doc(_userId)
        .set({
          'fromUserId': _userId,
          'fromName': myName,
          'createdAt': FieldValue.serverTimestamp(),
        });
  }

  @override
  Future<void> acceptRequest(String requestId) async {
    final reqDoc = await _requestsRef.doc(requestId).get();
    if (!reqDoc.exists) return;

    final data = reqDoc.data() as Map<String, dynamic>;
    final friendId = (data['fromUserId'] as String?) ?? requestId;
    final friendName = (data['fromName'] as String?) ?? 'Friend';

    final myDoc = await _firestore.collection('users').doc(_userId).get();
    final myData = myDoc.data() ?? {};
    final myName = (myData['displayName'] as String?) ?? 'Student';

    final batch = _firestore.batch();

    batch.set(_friendsRef.doc(friendId), {
      'displayName': friendName,
      'sharingLevel': 'busyOnly',
    });

    batch.set(
      _firestore
          .collection('users')
          .doc(friendId)
          .collection('friends')
          .doc(_userId),
      {'displayName': myName, 'sharingLevel': 'busyOnly'},
    );

    batch.delete(_requestsRef.doc(requestId));

    await batch.commit();
  }

  @override
  Future<void> declineRequest(String requestId) async {
    await _requestsRef.doc(requestId).delete();
  }

  @override
  Future<void> setSharingLevel(String friendId, SharingLevel level) async {
    await _friendsRef.doc(friendId).update({
      'sharingLevel': level == SharingLevel.full ? 'full' : 'busyOnly',
    });
  }

  @override
  Future<void> removeFriend(String friendId) async {
    final batch = _firestore.batch();

    batch.delete(_friendsRef.doc(friendId));

    batch.delete(
      _firestore
          .collection('users')
          .doc(friendId)
          .collection('friends')
          .doc(_userId),
    );

    await batch.commit();
  }
}

/// Live implementation of [FriendScheduleRepository] backed by Firebase Firestore.
class FirestoreFriendScheduleRepository implements FriendScheduleRepository {
  FirestoreFriendScheduleRepository(this._firestore);

  final FirebaseFirestore _firestore;

  @override
  Stream<List<ClassEntry>> watchFriendClasses(String friendId) {
    return _firestore
        .collection('users')
        .doc(friendId)
        .collection('classes')
        .snapshots()
        .map((snap) {
      return snap.docs.map((doc) {
        final data = doc.data();
        return ClassEntry(
          id: doc.id,
          name: (data['name'] as String?) ?? '',
          days: List<int>.from((data['days'] as List?) ?? []),
          startMinute: (data['startMinute'] as int?) ?? 0,
          endMinute: (data['endMinute'] as int?) ?? 0,
          colorValue: (data['colorValue'] as int?) ?? 0xFF2E7D6B,
          room: data['room'] as String?,
          reminderMinutes: data['reminderMinutes'] as int?,
          sharedClassId: data['sharedClassId'] as String?,
        );
      }).toList();
    });
  }

  @override
  Stream<List<BusySlot>> watchBusySlots(String friendId) {
    return watchFriendClasses(friendId).map((classes) {
      final slots = <BusySlot>[];
      for (final c in classes) {
        for (final day in c.days) {
          slots.add(
            BusySlot(
              day: day,
              startMinute: c.startMinute,
              endMinute: c.endMinute,
            ),
          );
        }
      }
      return slots;
    });
  }
}

class FriendsScreen extends ConsumerStatefulWidget {
  const FriendsScreen({super.key});

  @override
  ConsumerState<FriendsScreen> createState() => _FriendsScreenState();
}

class _FriendsScreenState extends ConsumerState<FriendsScreen> {
  final _code = TextEditingController();
  final _busy = <String>{};
  bool _sending = false;
  String? _error;

  @override
  void dispose() {
    _code.dispose();
    super.dispose();
  }

  /// Runs [action]. Returns a message if it failed, or null if it worked.
  Future<String?> _attempt(Future<void> Function() action) async {
    try {
      await action();
      return null;
    } on AppFailure catch (failure) {
      return failure.message;
    } catch (_) {
      return 'Something went wrong. Please try again.';
    }
  }

  /// Runs one action at a time per [key], then shows its result.
  Future<void> _act(
    String key,
    Future<void> Function() action, {
    String? success,
  }) async {
    if (_busy.contains(key)) {
      return;
    }
    final messenger = ScaffoldMessenger.of(context);
    setState(() => _busy.add(key));
    final failure = await _attempt(action);
    if (!mounted) {
      return;
    }
    setState(() => _busy.remove(key));
    final message = failure ?? success;
    if (message != null) {
      messenger.showSnackBar(SnackBar(content: Text(message)));
    }
  }

  Future<void> _copy(String code) async {
    final messenger = ScaffoldMessenger.of(context);
    await Clipboard.setData(ClipboardData(text: code));
    messenger.showSnackBar(const SnackBar(content: Text('Code copied')));
  }

  Future<void> _sendRequest(String? ownCode) async {
    final code = _code.text.trim().toUpperCase();
    if (code.isEmpty) {
      setState(() => _error = 'Enter your friendâ€™s code.');
      return;
    }
    if (code == ownCode?.toUpperCase()) {
      setState(() => _error = 'Thatâ€™s your own code.');
      return;
    }

    final repo = ref.read(friendRepositoryProvider);
    final messenger = ScaffoldMessenger.of(context);
    setState(() {
      _sending = true;
      _error = null;
    });
    final failure = await _attempt(() => repo.sendRequestByCode(code));
    if (!mounted) {
      return;
    }
    setState(() {
      _sending = false;
      _error = failure;
    });
    if (failure == null) {
      _code.clear();
      messenger.showSnackBar(const SnackBar(content: Text('Request sent')));
    }
  }

  Future<void> _togglePrivacy(Friend friend) async {
    final turningOn = friend.sharingLevel == SharingLevel.busyOnly;
    final first = friend.displayName.split(' ').first;

    if (turningOn) {
      final confirmed = await showDialog<bool>(
        context: context,
        builder: (dialogContext) => AlertDialog(
          title: Text('Show class names to ${friend.displayName}?'),
          content: Text(
            '$first will see the name, room, and time of every class in '
            'your timetable. You can turn this off any time.',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext, false),
              child: const Text('Cancel'),
            ),
            TextButton(
              onPressed: () => Navigator.pop(dialogContext, true),
              child: const Text('Show names'),
            ),
          ],
        ),
      );
      if (confirmed != true || !mounted) {
        return;
      }
    }

  ref,
  friendId,

    }
  }

/// Explains why reminders need permission, then asks for it.
///
/// Runs once per app session after sign-in. With [force], used by the button
/// on the Me screen, it runs even if the explanation was already shown.
Future<void> offerReminders(
  BuildContext context,
  WidgetRef ref, {
  bool force = false,
}) async {
  final offer = ref.read(reminderOfferProvider);
  if (offer.shown && !force) {
    return;
  }

  final scheduler = ref.read(reminderSchedulerProvider);
  if (await scheduler.isAllowed()) {
    return;
  }
  if (!context.mounted) {
    return;
  }
  offer.shown = true;

  final agreed = await showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: const Text('Allow reminders?'),
      content: const Text(
        'ClassSync needs notification access to remind you before each '
        'class. You can change this later in your phone settings.',
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(dialogContext, false),
          child: const Text('Not now'),
        ),
        TextButton(
          onPressed: () => Navigator.pop(dialogContext, true),
          child: const Text('Allow reminders'),
        ),
      ],
    ),
  );
  if (agreed != true || !context.mounted) {
    return;
  }

  final granted = await scheduler.requestPermission();
  if (!context.mounted) {
    return;
  }
  ref.invalidate(reminderAccessProvider);

  if (granted) {
    // Schedule right away instead of waiting for the next change.
    ref.invalidate(reminderSyncProvider);
    return;
  }

  // After two refusals Android stops showing its dialog, so point to settings.
  if (force) {
    await showDialog<void>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Reminders are off'),
        content: const Text(
          'To turn them on, open your phoneâ€™s Settings, then Apps, then '
          'ClassSync, then Notifications.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext),
            child: const Text('OK'),
          ),
        ],
      ),
    );
  }
}

  final scheduler = ref.watch(reminderSchedulerProvider);
  final auth = ref.watch(authStateProvider);

    now: DateTime.now(),

  ReminderScheduler scheduler,
  List<PlannedReminder> plan,

    // Nothing to show if the phone blocks notifications.

    debugPrint('Could not update reminders: $error');

import 'dart:math';

import 'package:flutter/material.dart';

class WeekGrid extends StatelessWidget {
  const WeekGrid({required this.classes, required this.onTapClass, super.key});

  final List<ClassEntry> classes;
  final ValueChanged<ClassEntry> onTapClass;

  // Minimum visible range: 7:00 to 22:00, in minutes since midnight.
  static const _minStart = 420;
  static const _minEnd = 1320;
  static const _laneHeight = 380.0;

  /// Earliest class start, rounded down to a whole hour.
  int get _start {
    final earliest = classes.fold<int>(
      _minStart,
      (value, c) => min(value, c.startMinute),
    );
    return earliest ~/ 60 * 60;
  }

  /// Latest class end, rounded up to a whole hour.
  int get _end {
    final latest = classes.fold<int>(
      _minEnd,
      (value, c) => max(value, c.endMinute),
    );
    return (latest + 59) ~/ 60 * 60;
  }

  double _position(int minutes) {
    final fraction = (minutes - _start) / (_end - _start);
    return max(0.0, min(1.0, fraction)) * _laneHeight;
  }

/// Returns a message for the first problem found, or null if the class is fine.
///
/// Runs before anything is saved so the user gets a clear message. The data
/// layer has to enforce the same rules on its side.
String? validateClass(
  ClassEntry draft, {
  required List<ClassEntry> existing,
  required int defaultReminderMinutes,
}) {
  if (draft.name.trim().isEmpty) return 'Add a class name.';
  if (draft.days.isEmpty) return 'Pick at least one day.';
  if (draft.startMinute < 0 || draft.endMinute > 1440) {
    return 'Times must be within one day.';
  }
  if (draft.endMinute <= draft.startMinute) {
    return 'End time must be after the start time.';
  }

  final lead = draft.reminderMinutes ?? defaultReminderMinutes;
  if (draft.startMinute - lead < 0) {
    return 'That reminder falls before midnight. Choose a shorter one.';
  }

  for (final other in existing) {
    // Editing a class must not clash with its own saved copy.
    if (other.id == draft.id) {
      continue;
    }
    final sharesDay = other.days.any(draft.days.contains);
    final overlaps =
        draft.startMinute < other.endMinute &&
        other.startMinute < draft.endMinute;
    if (sharesDay && overlaps) {
      return 'This overlaps with ${other.name}.';
    }
  }
  return null;
}

import '../core/utils/schedule_format.dart';
import '../domain/models/class_entry.dart';

/// How many days ahead reminders are scheduled. The app refreshes the window
/// every time it opens and whenever classes change.
const reminderWindowDays = 7;

/// iOS keeps at most 64 pending notifications, so we stay below that.
const maxScheduledReminders = 60;

class PlannedReminder {
  const PlannedReminder({
    required this.id,
    required this.at,
    required this.title,
    required this.body,
  });

  /// 0, 1, 2... in time order. Safe to reuse because every refresh cancels
  /// all pending reminders before scheduling the new set.
  final int id;

  /// Local time the reminder should appear.
  final DateTime at;
  final String title;
  final String body;
}

/// Works out which reminders should be scheduled from [now] onwards.
List<PlannedReminder> planReminders({
  required List<ClassEntry> classes,
  required int defaultReminderMinutes,
  required DateTime now,
  int days = reminderWindowDays,
  int maxCount = maxScheduledReminders,
}) {
  final found = <({DateTime at, ClassEntry entry})>[];

  for (var offset = 0; offset < days; offset++) {
    // DateTime rolls over month ends by itself, so day + offset is safe.
    final day = DateTime(now.year, now.month, now.day + offset);
    for (final entry in classes) {
      if (!entry.days.contains(day.weekday)) {
        continue;
      }
      final lead = entry.reminderMinutes ?? defaultReminderMinutes;
      final at = DateTime(
        day.year,
        day.month,
        day.day,
        0,
        entry.startMinute - lead,
      );
      if (at.isAfter(now)) {
        found.add((at: at, entry: entry));
      }
    }
  }

  found.sort((a, b) => a.at.compareTo(b.at));
  final chosen = found.take(maxCount).toList();

  return [
    for (var i = 0; i < chosen.length; i++)
      PlannedReminder(
        id: i,
        at: chosen[i].at,
        title: chosen[i].entry.name,
        body: _body(chosen[i].entry),
      ),
  ];
}

String _body(ClassEntry entry) {
  final room = entry.room;
  final start = 'Starts at ${formatMinutes(entry.startMinute)}';
  return room == null || room.isEmpty ? start : '$start Â· $room';
}

import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:timezone/timezone.dart' as tz;

import '../../logic/reminder_planner.dart';
import 'reminder_scheduler.dart';

class LocalNotificationScheduler implements ReminderScheduler {
  LocalNotificationScheduler([FlutterLocalNotificationsPlugin? plugin])
    : _plugin = plugin ?? FlutterLocalNotificationsPlugin();

  final FlutterLocalNotificationsPlugin _plugin;

  // Runs the plugin setup once, on first use.
  late final Future<void> _ready = _initialize();

  // Updates run one at a time, in the order they were requested.
  Future<void> _lastUpdate = Future<void>.value();

  static const _details = NotificationDetails(
    android: AndroidNotificationDetails(
      'class_reminders',
      'Class reminders',
      channelDescription: 'Reminders before your classes start',
      importance: Importance.high,
      priority: Priority.high,
    ),
  );

  AndroidFlutterLocalNotificationsPlugin? get _android => _plugin
      .resolvePlatformSpecificImplementation<
        AndroidFlutterLocalNotificationsPlugin
      >();

  Future<void> _initialize() async {
    await _plugin.initialize(
      settings: const InitializationSettings(
        android: AndroidInitializationSettings('@mipmap/ic_launcher'),
        iOS: DarwinInitializationSettings(
          requestAlertPermission: false,
          requestBadgePermission: false,
          requestSoundPermission: false,
        ),
      ),
    );
  }

  @override
  Future<bool> isAllowed() async {
    await _ready;
    final android = _android;
    // iOS permission handling is added with the iOS compile check.
    if (android == null) {
      return true;
    }
    return await android.areNotificationsEnabled() ?? false;
  }

  @override
  Future<bool> requestPermission() async {
    await _ready;
    final android = _android;
    if (android == null) {
      return true;
    }
    return await android.requestNotificationsPermission() ?? false;
  }

  @override
  Future<void> replaceAll(List<PlannedReminder> reminders) {
    final run = _lastUpdate.then(
      (_) => _replace(reminders),
      onError: (_) => _replace(reminders),
    );
    _lastUpdate = run;
    return run;
  }

  Future<void> _replace(List<PlannedReminder> reminders) async {
    await _ready;
    await _plugin.cancelAllPendingNotifications();
    if (reminders.isEmpty) {
      return;
    }

    // Exact times when Android allows it, otherwise close to the time.
    final exact = await _android?.canScheduleExactNotifications() ?? true;
    final mode = exact
        ? AndroidScheduleMode.exactAllowWhileIdle
        : AndroidScheduleMode.inexactAllowWhileIdle;

    for (final reminder in reminders) {
      try {
        await _plugin.zonedSchedule(
          id: reminder.id,
          title: reminder.title,
          body: reminder.body,
          // An exact moment in time, so the phone's time zone doesn't matter.
          scheduledDate: tz.TZDateTime.from(reminder.at.toUtc(), tz.UTC),
          notificationDetails: _details,
          androidScheduleMode: mode,
        );
      } on ArgumentError {
        // The time passed while we were scheduling. Skip it.
      }
    }
  }
}

/// What the app needs from the phone's notification system.

abstract interface class ReminderScheduler {

import 'package:classsync/logic/reminder_planner.dart';
import 'package:flutter_test/flutter_test.dart';

ClassEntry _class({
  String id = 'a',
  String name = 'Calculus',
  List<int> days = const [1, 3, 5],
  int start = 480,
  int? reminder,
  String? room,
}) {
  return ClassEntry(
    id: id,
    name: name,
    days: days,
    startMinute: start,
    endMinute: start + 60,
    colorValue: 0xFF2E7D6B,
    reminderMinutes: reminder,
    room: room,
  );
}

// Monday 19 October 2026, 07:00.
final _monday = DateTime(2026, 10, 19, 7);

List<PlannedReminder> _plan(
  List<ClassEntry> classes, {
  DateTime? now,
  int maxCount = 60,
}) {
  return planReminders(
    classes: classes,
    defaultReminderMinutes: 10,
    now: now ?? _monday,
    maxCount: maxCount,
  );
}

void main() {
  test('plans one reminder per class day in the next 7 days', () {
    final plan = _plan([_class()]);

    expect(plan.map((r) => r.at), [
      DateTime(2026, 10, 19, 7, 50),
      DateTime(2026, 10, 21, 7, 50),
      DateTime(2026, 10, 23, 7, 50),
    ]);
  });

  test('skips reminders that are already in the past', () {
    final plan = _plan([_class()], now: DateTime(2026, 10, 19, 8));

    expect(plan.length, 2);
    expect(plan.first.at, DateTime(2026, 10, 21, 7, 50));
  });

  test('a per-class reminder overrides the default', () {
    final plan = _plan([_class(reminder: 30)]);

    expect(plan.first.at, DateTime(2026, 10, 19, 7, 30));
  });

  test('reminders are sorted by time and numbered from zero', () {
    final plan = _plan([
      _class(id: 'a', name: 'Calculus', days: [1], start: 480),
      _class(id: 'b', name: 'Biology', days: [1], start: 450),
    ]);

    expect(plan.map((r) => r.title), ['Biology', 'Calculus']);
    expect(plan.map((r) => r.id), [0, 1]);
  });

  test('keeps only the earliest reminders when over the limit', () {
    final plan = _plan([_class()], maxCount: 2);

    expect(plan.length, 2);
    expect(plan.last.at, DateTime(2026, 10, 21, 7, 50));
  });

  test('the message shows the start time and the room', () {
    final plan = _plan([_class(room: 'Room 204')]);

    expect(plan.first.body, 'Starts at 8:00 Â· Room 204');
  });

  test('no classes means no reminders', () {
    expect(_plan([]), isEmpty);
  });
}

import 'package:classsync/core/providers/repository_providers.dart';
import 'package:classsync/features/reminders/reminder_providers.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';

import 'support/fake_reminder_scheduler.dart';

void main() {
  test('signing in schedules reminders and signing out clears them', () async {
    final scheduler = FakeReminderScheduler();
    final container = ProviderContainer(
      overrides: [reminderSchedulerProvider.overrideWithValue(scheduler)],
    );
    addTearDown(container.dispose);
    container.listen(reminderSyncProvider, (_, _) {});

    final auth = container.read(authRepositoryProvider);
    await auth.signInWithGoogle();
    await pumpEventQueue();
    expect(scheduler.lastPlan, isNotEmpty);

    await auth.signOut();
    await pumpEventQueue();
    expect(scheduler.lastPlan, isEmpty);
  });

  test('nothing is scheduled while notifications are blocked', () async {
    final scheduler = FakeReminderScheduler(allowed: false);
    final container = ProviderContainer(
      overrides: [reminderSchedulerProvider.overrideWithValue(scheduler)],
    );
    addTearDown(container.dispose);
    container.listen(reminderSyncProvider, (_, _) {});

    await container.read(authRepositoryProvider).signInWithGoogle();
    await pumpEventQueue();

    expect(scheduler.lastPlan, isNull);
  });
}

import 'package:classsync/logic/reminder_planner.dart';
import 'package:classsync/services/notifications/reminder_scheduler.dart';

/// Stands in for the phone's notification system in tests.
class FakeReminderScheduler implements ReminderScheduler {
  FakeReminderScheduler({this.allowed = true, this.grantOnRequest = true});

  bool allowed;
  final bool grantOnRequest;
  int requestCount = 0;
  List<PlannedReminder>? lastPlan;

  @override
  Future<bool> isAllowed() async => allowed;

  @override
  Future<bool> requestPermission() async {
    requestCount++;
    allowed = grantOnRequest;
    return allowed;
  }

  @override
  Future<void> replaceAll(List<PlannedReminder> reminders) async {
    lastPlan = reminders;
  }
}
